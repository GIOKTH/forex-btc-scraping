#property copyright "BTCUSD Smart Analysis Bot"

#include <Trade\Trade.mqh>
#include <Trade\PositionInfo.mqh>

//--- Input parameters
input double ProfitTargetDollar = 4.0;     // $4 profit target
input double StopLossDollar = 2.0;         // $2 stop loss
input int MaxDailyTrades = 20;             // 20 trades max
input double LotSize = 0.01;               // Fixed lot
input double MaxDailyLoss = 20.0;          // Max $20 loss per day
input bool EnableLogging = true;
input int MagicNumber = 666888;

//--- Smart Analysis Parameters
input group "=== MARKET ANALYSIS ==="
input int EMA_Fast = 12;                   // Fast EMA for trend
input int EMA_Slow = 26;                   // Slow EMA for trend
input int RSI_Period = 14;                 // RSI period
input int ATR_Period = 14;                 // ATR for volatility
input double MinTrendStrength = 0.05;       // Very low trend requirement
input double MinVolatility = 10;            // Very low volatility requirement
input bool OnlyTradeWithTrend = false;      // Trade in any direction
input bool AvoidChoppyMarkets = false;      // Allow some choppy markets

//--- Entry Conditions
input group "=== ENTRY RULES ==="
input double RSI_OversoldLevel = 40;        // Much more relaxed
input double RSI_OverboughtLevel = 60;     // Much more relaxed
input bool RequireVolumeConfirmation = true; // Wait for volume spike
input bool CheckPriceAction = false;        // Skip price action check
input int LookbackBars = 10;               // Bars to analyze for setup

//--- Global variables
string symbol = "BTCUSD";
CTrade trade;
CPositionInfo position;

// Indicators
int emaFastHandle, emaSlowHandle, rsiHandle, atrHandle;

// Position tracking
bool inPosition = false;
int dailyTrades = 0;
double dailyLoss = 0;
datetime currentDay = 0;
bool dailyLossLimitHit = false;

// Market analysis
struct MarketCondition
{
    double trendStrength;
    bool isUptrend;
    bool isDowntrend;
    bool isSideways;
    double volatility;
    bool goodVolatility;
    double rsi;
    bool marketSuitable;
    string analysis;
};

//+------------------------------------------------------------------+
//| Expert initialization function                                   |
//+------------------------------------------------------------------+
int OnInit()
{
    Print("=== SMART ANALYSIS BOT STARTED ===");
    Print("WILL ANALYZE MARKET BEFORE PLACING ORDERS!");
    Print("Target: $", ProfitTargetDollar, " | Stop: $", StopLossDollar);
    Print("Only trades when market conditions are favorable");
    
    // Select symbol
    if(!SymbolSelect(symbol, true))
    {
        Print("ERROR: Cannot select symbol ", symbol);
        return INIT_FAILED;
    }
    
    // Create indicators for smart analysis
    emaFastHandle = iMA(symbol, PERIOD_M5, EMA_Fast, 0, MODE_EMA, PRICE_CLOSE);
    emaSlowHandle = iMA(symbol, PERIOD_M5, EMA_Slow, 0, MODE_EMA, PRICE_CLOSE);
    rsiHandle = iRSI(symbol, PERIOD_M5, RSI_Period, PRICE_CLOSE);
    atrHandle = iATR(symbol, PERIOD_M5, ATR_Period);
    
    if(emaFastHandle == INVALID_HANDLE || emaSlowHandle == INVALID_HANDLE || 
       rsiHandle == INVALID_HANDLE || atrHandle == INVALID_HANDLE)
    {
        Print("ERROR: Cannot create indicators for analysis");
        return INIT_FAILED;
    }
    
    // Setup trading
    trade.SetExpertMagicNumber(MagicNumber);
    trade.SetMarginMode();
    trade.SetTypeFillingBySymbol(symbol);
    trade.SetDeviationInPoints(20);
    
    // Wait for indicators
    Sleep(3000);
    
    // Initialize tracking
    currentDay = TimeCurrent();
    dailyTrades = 0;
    dailyLoss = 0;
    
    Print("=== SMART BOT READY ===");
    Print("Will analyze: Trend, Volatility, RSI, Price Action");
    Print("Min trend strength: ", MinTrendStrength, "%");
    Print("Min volatility (ATR): ", MinVolatility);
    
    return INIT_SUCCEEDED;
}

//+------------------------------------------------------------------+
//| Expert deinitialization function                                 |
//+------------------------------------------------------------------+
void OnDeinit(const int reason)
{
    Print("=== SMART BOT STOPPED ===");
    Print("Daily trades: ", dailyTrades);
    Print("Daily loss: $", NormalizeDouble(dailyLoss, 2));
    
    // Release indicators
    if(emaFastHandle != INVALID_HANDLE) IndicatorRelease(emaFastHandle);
    if(emaSlowHandle != INVALID_HANDLE) IndicatorRelease(emaSlowHandle);
    if(rsiHandle != INVALID_HANDLE) IndicatorRelease(rsiHandle);
    if(atrHandle != INVALID_HANDLE) IndicatorRelease(atrHandle);
}

//+------------------------------------------------------------------+
//| Expert tick function                                             |
//+------------------------------------------------------------------+
void OnTick()
{
    // Check new day
    CheckNewDay();
    
    // Check safety limits
    if(!CheckSafetyLimits()) return;
    
    // Update position
    UpdatePosition();
    
    // Manage position if open
    if(inPosition)
    {
        ManagePosition();
        return;
    }
    
    // Check if we can trade
    if(!CanTrade()) return;
    
    // SMART MARKET ANALYSIS BEFORE TRADING
    AnalyzeMarketAndTrade();
}

//+------------------------------------------------------------------+
//| Check for new day                                               |
//+------------------------------------------------------------------+
void CheckNewDay()
{
    datetime now = TimeCurrent();
    MqlDateTime nowStruct, currentDayStruct;
    
    TimeToStruct(now, nowStruct);
    TimeToStruct(currentDay, currentDayStruct);
    
    if(nowStruct.day != currentDayStruct.day)
    {
        Print("=== NEW TRADING DAY ===");
        Print("Yesterday: ", dailyTrades, " trades, $", NormalizeDouble(dailyLoss, 2), " loss");
        
        dailyTrades = 0;
        dailyLoss = 0;
        currentDay = now;
        dailyLossLimitHit = false;
    }
}

//+------------------------------------------------------------------+
//| Check safety limits                                             |
//+------------------------------------------------------------------+
bool CheckSafetyLimits()
{
    if(dailyLossLimitHit || dailyLoss >= MaxDailyLoss)
    {
        if(!dailyLossLimitHit)
        {
            Print("DAILY LOSS LIMIT HIT: $", NormalizeDouble(dailyLoss, 2));
            dailyLossLimitHit = true;
        }
        return false;
    }
    return true;
}

//+------------------------------------------------------------------+
//| Update position status                                           |
//+------------------------------------------------------------------+
void UpdatePosition()
{
    inPosition = false;
    for(int i = 0; i < PositionsTotal(); i++)
    {
        if(position.SelectByIndex(i))
        {
            if(position.Symbol() == symbol && position.Magic() == MagicNumber)
            {
                inPosition = true;
                break;
            }
        }
    }
}

//+------------------------------------------------------------------+
//| Manage position                                                  |
//+------------------------------------------------------------------+
void ManagePosition()
{
    if(!position.Select(symbol)) return;
    
    double profit = position.Profit() + position.Swap() + position.Commission();
    
    if(profit >= ProfitTargetDollar)
    {
        Print("PROFIT TARGET HIT: $", NormalizeDouble(profit, 2));
        ClosePosition("Profit Target");
    }
    else if(profit <= -StopLossDollar)
    {
        Print("STOP LOSS HIT: $", NormalizeDouble(profit, 2));
        ClosePosition("Stop Loss");
    }
}

//+------------------------------------------------------------------+
//| Close position                                                   |
//+------------------------------------------------------------------+
void ClosePosition(string reason)
{
    if(position.Select(symbol))
    {
        double profit = position.Profit() + position.Swap() + position.Commission();
        
        if(trade.PositionClose(position.Ticket()))
        {
            if(profit < 0)
            {
                dailyLoss += MathAbs(profit);
            }
            
            inPosition = false;
            
            Print("Position closed (", reason, "). P&L: $", NormalizeDouble(profit, 2));
            Print("Daily loss total: $", NormalizeDouble(dailyLoss, 2));
        }
    }
}

//+------------------------------------------------------------------+
//| Check if we can trade                                           |
//+------------------------------------------------------------------+
bool CanTrade()
{
    if(dailyTrades >= MaxDailyTrades) return false;
    if(Bars(symbol, PERIOD_M5) < EMA_Slow + LookbackBars) return false;
    return true;
}

//+------------------------------------------------------------------+
//| SMART MARKET ANALYSIS BEFORE TRADING                            |
//+------------------------------------------------------------------+
void AnalyzeMarketAndTrade()
{
    // Get all indicator values
    double emaFast = GetIndicatorValue(emaFastHandle, 0);
    double emaSlow = GetIndicatorValue(emaSlowHandle, 0);
    double rsi = GetIndicatorValue(rsiHandle, 0);
    double atr = GetIndicatorValue(atrHandle, 0);
    
    if(emaFast == 0 || emaSlow == 0 || rsi == 0 || atr == 0)
    {
        Print("Waiting for all indicators...");
        return;
    }
    
    // Perform comprehensive market analysis
    MarketCondition market = AnalyzeMarketConditions(emaFast, emaSlow, rsi, atr);
    
    // Log market analysis every 5 minutes
    static datetime lastAnalysisLog = 0;
    if(TimeCurrent() - lastAnalysisLog >= 300)
    {
        Print("=== MARKET ANALYSIS ===");
        Print("Trend Strength: ", NormalizeDouble(market.trendStrength, 3), "%");
        Print("Trend Direction: ", market.isUptrend ? "UP" : (market.isDowntrend ? "DOWN" : "SIDEWAYS"));
        Print("Volatility (ATR): ", NormalizeDouble(market.volatility, 2));
        Print("RSI: ", NormalizeDouble(market.rsi, 1));
        Print("Market Suitable: ", market.marketSuitable ? "YES" : "NO");
        Print("Analysis: ", market.analysis);
        lastAnalysisLog = TimeCurrent();
    }
    
    // Only trade if market conditions are favorable
    if(!market.marketSuitable)
    {
        if(EnableLogging)
        {
            Print("Waiting for better market conditions: ", market.analysis);
        }
        return;
    }
    
    // Look for high-quality setups only
    CheckForQualitySetups(market);
}

//+------------------------------------------------------------------+
//| Analyze market conditions comprehensively                       |
//+------------------------------------------------------------------+
MarketCondition AnalyzeMarketConditions(double emaFast, double emaSlow, double rsi, double atr)
{
    MarketCondition market;
    
    // Calculate trend strength
    market.trendStrength = MathAbs(emaFast - emaSlow) / emaSlow * 100;
    
    // Determine trend direction
    market.isUptrend = (emaFast > emaSlow) && (market.trendStrength >= MinTrendStrength);
    market.isDowntrend = (emaFast < emaSlow) && (market.trendStrength >= MinTrendStrength);
    market.isSideways = market.trendStrength < MinTrendStrength;
    
    // Check volatility
    market.volatility = atr;
    market.goodVolatility = atr >= MinVolatility;
    market.rsi = rsi;
    
    // Analyze price action
    bool goodPriceAction = AnalyzePriceAction();
    
    // Determine if market is suitable for trading
    market.marketSuitable = true;
    market.analysis = "Good conditions";
    
    // Check each condition - MUCH MORE RELAXED
    if(market.isSideways && AvoidChoppyMarkets && market.trendStrength < 0.02)
    {
        market.marketSuitable = false;
        market.analysis = "Market extremely flat (trend: " + DoubleToString(market.trendStrength, 4) + "%)";
    }
    else if(!market.goodVolatility)
    {
        market.marketSuitable = false;
        market.analysis = "Volatility too low (ATR: " + DoubleToString(atr, 1) + ")";
    }
    else if(rsi > 45 && rsi < 55)
    {
        market.marketSuitable = false;
        market.analysis = "RSI too neutral (" + DoubleToString(rsi, 1) + ") - need > 60 or < 40";
    }
    
    return market;
}

//+------------------------------------------------------------------+
//| Analyze recent price action for quality                         |
//+------------------------------------------------------------------+
bool AnalyzePriceAction()
{
    if(!CheckPriceAction) return true;
    
    double highs[], lows[], closes[];
    ArraySetAsSeries(highs, true);
    ArraySetAsSeries(lows, true);
    ArraySetAsSeries(closes, true);
    
    if(CopyHigh(symbol, PERIOD_M5, 0, LookbackBars, highs) < LookbackBars ||
       CopyLow(symbol, PERIOD_M5, 0, LookbackBars, lows) < LookbackBars ||
       CopyClose(symbol, PERIOD_M5, 0, LookbackBars, closes) < LookbackBars)
    {
        return false;
    }
    
    // Check for clear directional movement
    double rangeHigh = highs[ArrayMaximum(highs)];
    double rangeLow = lows[ArrayMinimum(lows)];
    double totalRange = rangeHigh - rangeLow;
    
    // Check recent momentum
    double recentMove = MathAbs(closes[0] - closes[5]);
    double momentum = recentMove / totalRange;
    
    // Good price action if recent move is significant
    return momentum > 0.3;
}

//+------------------------------------------------------------------+
//| Check for quality setups only                                   |
//+------------------------------------------------------------------+
void CheckForQualitySetups(MarketCondition &market)
{
    double currentPrice = SymbolInfoDouble(symbol, SYMBOL_BID);
    double emaFast = GetIndicatorValue(emaFastHandle, 0);
    double emaSlow = GetIndicatorValue(emaSlowHandle, 0);
    
    bool qualityBuySetup = false;
    bool qualitySellSetup = false;
    string setupReason = "";
    
    // QUALITY BUY SETUP - MUCH MORE RELAXED
    if(market.rsi <= RSI_OversoldLevel && market.marketSuitable)
    {
        qualityBuySetup = true;
        setupReason = "RSI_Oversold_" + DoubleToString(market.rsi, 1);
    }
    
    // QUALITY SELL SETUP - MUCH MORE RELAXED
    if(market.rsi >= RSI_OverboughtLevel && market.marketSuitable)
    {
        qualitySellSetup = true;
        setupReason = "RSI_Overbought_" + DoubleToString(market.rsi, 1);
    }
    
    // Execute only quality setups
    if(qualityBuySetup && !qualitySellSetup)
    {
        Print("=== QUALITY BUY SETUP DETECTED ===");
        Print("RSI: ", NormalizeDouble(market.rsi, 1), " (oversold)");
        Print("Trend: ", market.isUptrend ? "UP" : (market.isDowntrend ? "DOWN" : "SIDEWAYS"));
        Print("Volatility: ", NormalizeDouble(market.volatility, 2));
        ExecuteQualityTrade(ORDER_TYPE_BUY, setupReason);
    }
    else if(qualitySellSetup && !qualityBuySetup)
    {
        Print("=== QUALITY SELL SETUP DETECTED ===");
        Print("RSI: ", NormalizeDouble(market.rsi, 1), " (overbought)");
        Print("Trend: ", market.isUptrend ? "UP" : (market.isDowntrend ? "DOWN" : "SIDEWAYS"));
        Print("Volatility: ", NormalizeDouble(market.volatility, 2));
        ExecuteQualityTrade(ORDER_TYPE_SELL, setupReason);
    }
}

//+------------------------------------------------------------------+
//| Execute quality trade after analysis                            |
//+------------------------------------------------------------------+
void ExecuteQualityTrade(ENUM_ORDER_TYPE orderType, string reason)
{
    double lotSize = LotSize;
    
    // Validate lot size
    double minLot = SymbolInfoDouble(symbol, SYMBOL_VOLUME_MIN);
    if(lotSize < minLot) lotSize = minLot;
    
    string comment = "Quality_" + reason;
    
    bool success = false;
    
    if(orderType == ORDER_TYPE_BUY)
    {
        success = trade.Buy(lotSize, symbol, 0, 0, 0, comment);
    }
    else
    {
        success = trade.Sell(lotSize, symbol, 0, 0, 0, comment);
    }
    
    if(success)
    {
        dailyTrades++;
        inPosition = true;
        
        Print("=== QUALITY TRADE EXECUTED #", dailyTrades, " ===");
        Print("Type: ", (orderType == ORDER_TYPE_BUY) ? "BUY" : "SELL");
        Print("Setup: ", reason);
        Print("Analysis passed all quality checks!");
        Print("Target: $", ProfitTargetDollar, " | Stop: $", StopLossDollar);
        Print("Remaining: ", MaxDailyTrades - dailyTrades);
    }
    else
    {
        Print("Quality trade failed! Error: ", trade.ResultRetcode());
    }
}

//+------------------------------------------------------------------+
//| Get indicator value safely                                       |
//+------------------------------------------------------------------+
double GetIndicatorValue(int handle, int index)
{
    double buffer[];
    ArraySetAsSeries(buffer, true);
    
    if(CopyBuffer(handle, 0, index, 1, buffer) <= 0)
        return 0;
    
    return buffer[0];
}
